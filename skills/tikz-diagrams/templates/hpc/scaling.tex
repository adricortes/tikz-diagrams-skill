% Strong and weak scaling. Left: speedup S(p) = T(1)/T(p) for a fixed problem size,
% with the ideal line and Amdahl's law S(p) = 1/(f + (1-f)/p).
% Right: weak scaling efficiency E(p) = T(1)/T(p) with the work per process fixed.
% The measured points are illustrative. Replace them with your timings.
\documentclass[border=5pt]{standalone}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.18}
\pgfmathsetmacro{\fserial}{0.02} % serial fraction for the Amdahl curve
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
    group style={group size=2 by 1, horizontal sep=2.2cm},
    width=6.5cm, height=5.5cm,
    xmode=log, log basis x=2,
    xmin=1, xmax=256,
    xtick={1,4,16,64,256}, xticklabels={1,4,16,64,256},
    xlabel={Processes $p$},
    grid=major, grid style={gray!25},
    legend style={font=\footnotesize, draw=none, fill=white, fill opacity=0.8, text opacity=1, row sep=2pt},
]
% Strong scaling
\nextgroupplot[
    ymode=log, log basis y=2,
    ymin=1, ymax=256,
    ytick={1,4,16,64,256}, yticklabels={1,4,16,64,256},
    ylabel={Speedup $S(p)$},
    legend pos=north west,
    title={Strong scaling},
]
\addplot[dashed, black, domain=1:256] {x};
\addlegendentry{ideal}
\addplot[thick, gray, domain=1:256, samples=60] {1/(\fserial + (1-\fserial)/x)};
\addlegendentry{Amdahl, $f=\fserial$}
\addplot[thick, blue!70!black, mark=*] coordinates {(1,1) (2,1.95) (4,3.7) (8,6.8) (16,11.8) (32,18.6) (64,26.0) (128,31.5) (256,33.0)};
\addlegendentry{measured}
% Weak scaling
\nextgroupplot[
    ymin=0, ymax=1.1,
    ylabel={Efficiency $E(p)$},
    legend pos=south west,
    title={Weak scaling},
]
\addplot[dashed, black, domain=1:256] {1};
\addlegendentry{ideal}
\addplot[thick, blue!70!black, mark=*] coordinates {(1,1) (2,0.98) (4,0.96) (8,0.95) (16,0.93) (32,0.91) (64,0.88) (128,0.85) (256,0.81)};
\addlegendentry{measured}
\end{groupplot}
\end{tikzpicture}
\end{document}
